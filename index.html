🔗 Google Sheet 自動寫入設定 (Apps Script Webhook)

將部署好的 Google Apps Script 網頁應用程式網址貼在下方，學生登入（含學號與禮物領取狀態/時間點）與存檔 TPR 時即會自動寫入試算表：

📋 Google Apps Script 接收程式碼範本（含學號登入、禮物領取時間點與 TPR 紀錄）：

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date(),
    data.action || "",
    data.studentId || "",
    data.studentName || "",
    data.giftStatus || data.summaryStatus || "",
    data.giftAName + ":" + (data.giftAClaimed || "") + "(" + (data.giftATime || "") + ")",
    data.giftBName + ":" + (data.giftBClaimed || "") + "(" + (data.giftBTime || "") + ")",
    data.patientBed ? (data.patientBed + " " + data.patientName) : "",
    data.temp ? (data.temp + " (" + data.tempSite + ")") : "",
    data.pulse ? (data.pulse + " (" + data.pulseSite + ")") : "",
    data.resp || "",
    data.bp || ""
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}
